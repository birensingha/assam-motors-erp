        $am208Kind=is_string($am208KindRaw)

            ? $am208KindRaw : 'missed_checkout';
        $am208FromRaw=request()->query('am208_from','');
        $am208ToRaw=request()->query('am208_to','');
        $am208ValidDate=static function($v): string{
            if(!is_string($v)||!preg_match('/^\d{4}-\d{2}-\d{2}$/',$v))return '';
            [$y,$m,$d]=array_map('intval',explode('-',$v));
            return checkdate($m,$d,$y)?$v:'';
        };

        $am208To=$am208ValidDate($am208ToRaw);
        $am208Pattern='%'.str_replace(['\\','%','_'],['\\\\','\\%','\\_'],$am208Staff).'%';

            ->where('a.date','<',$am208Today)
            ->whereNotNull('a.check_in')->whereNull('a.check_out');
        // Badge always reflects complete unresolved backlog, not just filtered rows.
        $am130PendingCount=(clone $am130Base)->count();
        $am130Filtered=(clone $am130Base)
            ->leftJoin('staff as s','s.id','=','a.staff_id');
        if($am208Staff!==''){
            $am130Filtered->where(function($q)use($am208Pattern){
                $q->where('s.name','like',$am208Pattern)
                  ->orWhere('s.staff_code','like',$am208Pattern);
            });
        }
        if($am208From!=='')$am130Filtered->where('a.date','>=',$am208From);
        if($am208To!=='')$am130Filtered->where('a.date','<=',$am208To);

        $am208CheckoutMatched=$am208ShowMissing?(clone $am130Filtered)->count():0;
        $am130PendingRows=$am208ShowMissing
            ? (clone $am130Filtered)
                ->select('a.id','a.staff_id','a.date','a.check_in',
                    's.name as employee_name','s.staff_code','s.department')
                ->orderBy('a.date')->orderBy('a.id')->limit(100)->get()
            : collect();

        // declaration of absence, missing attendance or a disciplinary flag.
        $am208ShowToday=in_array($am208Kind,['all','no_checkin_today'],true);
        $am208NoCheckinRows=collect();
        $am208NoCheckinCount=0;

            && ($am208To===''||$am208To>=$am208Today);
        if($am208ShowToday && $am208TodayWithinRange){

                        ->whereColumn('a2.staff_id','s.id')

                        ->whereNotNull('a2.check_in');
                });
            if($am208Staff!==''){
                $am208TodayBase->where(function($q)use($am208Pattern){
                    $q->where('s.name','like',$am208Pattern)
